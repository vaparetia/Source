blade_flow.c
	デモ用剣軌跡エフェクト。
	ゲーム中の剣軌跡をデモ用に切り出したもの。
	剣のマトリクスに追従すると１フレーム遅れてしまうため
	毎フレーム、剣種類による剣先と剣根元のオフセットと手のマトリクスから
	剣先と剣根元の絶対座標を算出し、前フレームの両座標と速度ベクトルから
	フレーム間を補完し、ポリゴンを配置していく。
	軌跡のフェードイン・フェードアウトはメッセージでデモ側に制御してもらっている。

	＜呼び出し関数＞
	void *NewDemoBladeFlow( int name, FMATRIX *root, int bld_mode, int n_disp )
		int			name		: メッセージ受信用名前
		FMATRIX		*root		: 関節マトリクス
		int			bld_mode	: 剣種類
		int			n_disp		: 軌跡長さパラメータ

	ソリダスに最近接している炎座標と、ライデンに最近接している炎座標を選択し、
	その点の炎の強さに応じたテンプライトを設置する。
	また、ライデンとの最近接距離が一定以内の時に体燃えエフェクトを呼び出している。
	ダッシュ炎キャラがいなくなるとこのキャラも開放されるが
	再びダッシュ炎が発生した時に生成され、ダッシュ炎キャラがなくなるまで生き続ける。

	＜呼び出し関数＞
	void *NewSolidusDashFireLight( void )

	＜テンプライト登録関数＞
	void SetSolDashFireLightPos( FVECTOR *sol_lit_pos, FVECTOR *rai_lit_pos,
			     				float sol_lit_len2, float rai_lit_len2, float sol_lit_str, float rai_lit_str,
			     				int bodyfire_flag )
		FVECTOR		*sol_lit_pos	: ダッシュ炎のソリダスとの最近接点座標
		FVECTOR		*rai_lit_pos	: ダッシュ炎のライデンとの最近接点座標
		float		sol_lit_len2	: ソリダスと上の最近接点との距離の二乗
		float		rai_lit_len2	: ライデンと上の最近接点との距離の二乗
		float		sol_lit_str		: 上のソリダス最近接点の炎の強さ
		float		rai_lit_str		: 上のライデン最近接点の炎の強さ
		int			bodyfire_flag	: 体に炎着くかフラグ

	ソリダスのシュラウド燃えエフェクト。
