ステージw20cのファットマン戦のファットマンキャラ用のソースコード群

基本システムはsource/user/korekado/npc以下のNPCシステムを用いている。

fatman.h : ファットマン関連ヘッダ
    構造体の宣言やマクロ,定数群が記述されている。コメント参照
fatman.c : ファットマンキャラのActor処理関連
    ファットマンキャラの呼出、初期化、動作関数、破棄関数がある。

        FatmanMain( Work* work );
            ファットマンのメイン処理を行なう

        FAT_ActControl( NPCWORK *npc );
            NPC_ActControlのカスタマイズ版
            前の座標、移動量を取得する為

        Act( Work* work);
            GV_ACTORの動作関数
            NPCシステムの処理とファットマンのメイン関数コールをしている

        Die( Work* work);
            GV_ACTORの破棄関数
            後始末

        GetResource( Work* work,                // ファットマンワーク
                     int   name,                // GCLより取得する名前ID
                     int   where);              // GCLより取得するマップID
            初期設定関数
            各種初期化関数のコール＆メンバの設定を行なう

        NewFatman( int   name,                  // GCLより取得する名前ID
                   int   where);                // GCLより取得するマップID    
            GCLからコールされるファットマン生成関数

        NewCmdFatInvisible( void );
            GCL command でファットマンを非表示にする関数。
            デモでファットマンを消したい時に使用

        NewCmdFatVisible( void );
            GCL command でファットマンを表示にする関数

fatsound.c : 効果音＆ストリーミング関連
        FAT_SeManager( Work*    work);                    
            効果音マネージャ 内蔵SEを鳴らす間隔を設定

        FAT_CallSe( Work*    work,        // ファットマンワーク
                    int      nSeIndex,    // SE番号（インデックス）
                    FVECTOR* pvecPos,     // 発生位置
                    int      nMode);      // 再生のモード
            内蔵SEを鳴らす。ストリーミングとの兼ね合いなども判定して鳴らすかどうかきめる

        FAT_Say( Work*    work,         // ファットマンワーク
                 int      nSeIndex,     // SE番号（インデックス）
                 int      nMode)        // 再生のモード
            内蔵SEをファットマンの声として再生する。
            FAT_CallSeの発生位置をファットマンの頭の位置にしたもの

        FAT_StreamRequest( Work* work,           // ファットマンワーク
                           int   nIndex);        // ストリームテーブルインデックス
            ストリーミングリクエストに指定されたインデックスを代入して再生をさせる。
            ストリーミングが再生中ならその再生が終るのを待ってから再生する。
            実際に再生命令をだすのはFAT_StreamUpdate内

        FAT_StreamRequestJustTime( Work* work,    // ファットマンワーク
                                   int   nIndex); // ストリームテーブルインデックス
            ストリーミングリクエストに指定されたインデックスを代入して再生をさせる。
            ストリーミングが再生中ならあきらめる
            実際に再生命令をだすのはFAT_StreamUpdate内

        FAT_StreamRequestWithProc( Work* work,    // ファットマンワーク
                                   int   nIndex,  // ストリームテーブルインデックス
                                   int   nProc);  // 再生終了時に呼びたいプロシージャ
            ストリーミングリクエストに指定されたインデックスを代入して再生をさせる。
            再生終了時にnProcをコールする。
            ストリーミングが再生中ならその再生が終るのを待ってから再生する。
            実際に再生命令をだすのはFAT_StreamUpdate内

        FAT_StreamUpdate( Work* work );
            ストリーミングマネージャ。
            リクエストにあるインデックスが有効ならストリーミング再生命令を出す。

        FAT_DamageVoice( Work*  work,             // ファットマンワーク
                         int    weapon,           // 武器種類
                         int    part,             // ダメージを受けた部位
                         int    bNoDmg);          // ノーダメージフラグ
            ダメージを受けた時の内蔵SE音声呼びわけを武器の種類や部位から判定してコールする関数

        FAT_ResidentRollerSe( Work* work );
            常駐ローラー音。
            ファットマンのモーション再生状態をチェックして走行中ならSEを定期的にコールする

            位置からエリア番号を返す。

        FAT_GetContArea( int nArea);        // 判定基準となるエリア
            nAreaにたいして対となるエリアをエリア番号で返す

        FAT_GetContAreaBit( int nArea);     // 判定基準となるエリア
            nAreaにたいして対となるエリアをビットフラグ形式で返す

        FAT_GetContRunPutArea( Work* work, int nArea); 
            // 未使用
            対となる走り設置エリアを返す

        FAT_GetZoneFromInnerMin( NPCWORK* npc,          // NPCワーク
                                 int      nZoneAddr,    // 対象ゾーンアドレス
                                 FVECTOR* pfvec,        // 方向ベクトル

        FAT_GetZoneFromInnerMax( NPCWORK* npc,          // NPCワーク
                                 int      nZoneAddr,    // 対象ゾーンアドレス
                                 FVECTOR* pfvec,        // 方向ベクトル
                                 float*   pfRes)        // 内積結果が入る
            nZoneAddrの隣接ゾーンのなかからpfvecに対して最も内積値の大きいゾーンを返す

        GetFarZoneDir( NPCWORK* npc,            // NPCワーク
                       int      nAddr,          // 基準となるアドレス

                       int*     pnResDist)      // 結果距離
            指定方向にあって指定距離以上離れているゾーン取得

        GetFarZonePlayerDir(NPCWORK*    npc,            // NPCワーク

                            int*        pnResDist)      // 結果距離

        FAT_GetEvadeVector( Work*       work,           // ファットマンワーク

        FAT_FarZoneInFan( Work* work,       // ファットマンワーク
                          int*  pnDist,     // 結果距離
