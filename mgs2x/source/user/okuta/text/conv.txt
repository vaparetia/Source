            // 3点ベジェ補間
            MAO_MakeBezierMatrix( FMATRIX* pmatBezier,  	// 結果
		                  FVECTOR* pvec1,  		// 始点
                                  FVECTOR* pvec2,  		// 制御点
                                  FVECTOR* pvec3); 		// 終点
                始点,制御点,終点の情報からベジェ補間変換マトリクスを作成する。
                補間自体はMAO_Bezier3InterpQuickでおこなわれる。
		             |  1 -2  1  0 ||   pvec1 |
		pmatBezier = | -2  2  0  0 ||   pvec2 |
		             |  1  0  0  0 ||   pvec3 |
 	                     |  0  0  0  1 || 0 0 0 1 |

	    MAO_Bezier3InterpQuick( FVECTOR* pvecRes, 		// 結果
                                    FMATRIX* pmatBezier,  	// 変換マトリクス
                                    float    fRate); 		// 割合(0.f〜1.f)
                MAO_MakeBezierMatrixで作成した変換マトリクスを用いてベジェ補間を行なう。
                pvecRes = pmatBazier * vec( fRate^2, fRate, 1.f, 0.f );

	    MAO_Bezier3Interp( FVECTOR* pvecRes,		// 結果
                               FVECTOR* pvec1, 			// 始点
                               FVECTOR* pvec2, 			// 制御点
                               FVECTOR* pvec3, 			// 終点
                               float    fRate);		 	// 割合(0.f〜1.f)
                始点,制御点,終点の情報からベジェ補間変換マトリクスを作成,その後そのマトリクスを用いてベジェ補間を行なう
                pvecRes = ( 1.f - fRate )^2 * pvec1 + 2 * fRate * (1.f - fRate) * pvec2 + fRate^2 * pvec3
 	                  |  1 -2  1  0 ||   pvec1 || fRate^2 fRate 1 0 |
                pvecRes = | -2  2  0  0 ||   pvec2 |
 	                  |  1  0  0  0 ||   pvec3 |
 	                  |  0  0  0  1 || 0 0 0 1 |

memmot.c : メモリモーション関数群
    始めにモーションデータを全フレーム分メモリ上に展開してそれを参照することで処理時間を軽減するシステム。
    今回はカモメのみの使用になったので特化している(16関節など)

    struct MEMMOT_DATA : メモリモーションの展開したデータを扱う構造体
    struct MEMMOT_CTRL : 再生用の制御構造体
                
        MEMMOT_GetAbsRot( MEMMOT_CTRL*);
            モーションデータの中から現在再生しているモーションの再生時間にあった回転データの先頭を返す                

        MEMMOT_SetMotion( MEMMOT_CTRL*, int, int);
            再生するモーションを切替える

        MEMMOT_SetMotionSpeed( MEMMOT_CTRL*, float);
            再生速度を変更(未使用) : 整数速度指定しかまともに再生できない

        MEMMOT_AdjustMotion( MEMMOT_CTRL *mmt_ctrl);
            アジャスト機能
            
        MEMMOT_QuatSlerp( FVECTOR *res, FVECTOR *from, FVECTOR *to, float t )

        MEMMOT_ActMotion( MEMMOT_CTRL* mmt_ctrl);

        MEMMOT_ActMotion2( MOTION_CONTROL*, DG_OBJS*, FVECTOR* );

        MEMMOT_CheckObject_IsEnd( MEMMOT_CTRL*, int);
